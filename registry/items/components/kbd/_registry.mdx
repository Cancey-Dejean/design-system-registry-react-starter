Use `Kbd` for single keys and `KbdGroup` for shortcuts.

```tsx
import { Kbd, KbdGroup } from "@/components/kbd/kbd";

export function Example() {
  return (
    <>
      <Kbd>⌘</Kbd>

      <KbdGroup>
        <Kbd>⌘</Kbd>
        <Kbd>K</Kbd>
      </KbdGroup>
    </>
  );
}
```

Styles adapt inside tooltips via `in-data-[slot=tooltip-content]` rules.
