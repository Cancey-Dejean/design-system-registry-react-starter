Use a menubar for app chrome menus (File, Edit, View). Built on
[Base UI Menubar](https://base-ui.com/react/components/menubar) and dropdown menu primitives.

```tsx
import {
  Menubar,
  MenubarContent,
  MenubarItem,
  MenubarMenu,
  MenubarTrigger,
} from "@/components/menubar/menubar";

export function Example() {
  return (
    <Menubar>
      <MenubarMenu>
        <MenubarTrigger>File</MenubarTrigger>
        <MenubarContent>
          <MenubarItem>New Tab</MenubarItem>
          <MenubarItem>New Window</MenubarItem>
        </MenubarContent>
      </MenubarMenu>
    </Menubar>
  );
}
```

Compose checkboxes, radio groups, submenus, separators, and shortcuts like other menu components.
